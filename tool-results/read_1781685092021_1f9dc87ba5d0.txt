    75→    75→// ============================================
    76→    76→function useAnimatedCounter(target: number, duration: number = 2000) {
    77→    77→  const [count, setCount] = useState(0);
    78→    78→  const ref = useRef<HTMLDivElement>(null);
    79→    79→  const isInView = useInView(ref, { once: true, margin: '-50px' });
    80→    80→  const hasAnimated = useRef(false);
    81→    81→
    82→    82→  useEffect(() => {
    83→    83→    if (!isInView || hasAnimated.current) return;
    84→    84→    hasAnimated.current = true;
    85→    85→    const startTime = Date.now();
    86→    86→    const step = () => {
    87→    87→      const elapsed = Date.now() - startTime;
    88→    88→      const progress = Math.min(elapsed / duration, 1);
    89→    89→      const eased = 1 - Math.pow(1 - progress, 3);
    90→    90→      setCount(Math.floor(eased * target));
    91→    91→      if (progress < 1) requestAnimationFrame(step);
    92→    92→    };
    93→    93→    requestAnimationFrame(step);
    94→    94→  }, [isInView, target, duration]);
    95→    95→
    96→    96→  return { count, ref };
    97→    97→}
