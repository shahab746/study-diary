    54→function getDateLine(): string {
    55→  const days = ['SUNDAY', 'MONDAY', 'TUESDAY', 'WEDNESDAY', 'THURSDAY', 'FRIDAY', 'SATURDAY'];
    56→  const months = ['JANUARY', 'FEBRUARY', 'MARCH', 'APRIL', 'MAY', 'JUNE', 'JULY', 'AUGUST', 'SEPTEMBER', 'OCTOBER', 'NOVEMBER', 'DECEMBER'];

   229→/** Circular progress ring SVG component */
   230→function ProgressRing({
   231→  size = 68,
   232→  strokeWidth = 5,
   233→  progress = 0,
   234→  color = '#FF3B30',
   235→  trackColor = 'rgba(255,255,255,0.06)',
   236→  children,
   237→}: {
   238→  size?: number;
   239→  strokeWidth?: number;
   240→  progress?: number;
   241→  color?: string;
   242→  trackColor?: string;
   243→  children?: React.ReactNode;
   244→}) {
   245→  const radius = (size - strokeWidth) / 2;
   246→  const circumference = 2 * Math.PI * radius;
   247→  const offset = circumference * (1 - Math.min(progress, 100) / 100);
   248→
   249→  return (
   250→    <div style={{ position: 'relative', width: size, height: size, flexShrink: 0 }}>
   251→      <svg width={size} height={size} style={{ transform: 'rotate(-90deg)' }}>
   252→        <circle
   253→          cx={size / 2} cy={size / 2} r={radius}
   254→          fill="none" stroke={trackColor} strokeWidth={strokeWidth}
   255→        />
   256→        <motion.circle
   257→          cx={size / 2} cy={size / 2} r={radius}
   258→          fill="none" stroke={color} strokeWidth={strokeWidth}
   259→          strokeLinecap="round"
   260→          strokeDasharray={circumference}
   261→          initial={{ strokeDashoffset: circumference }}
   262→          animate={{ strokeDashoffset: offset }}
   263→          transition={{ duration: 1.2, ease: [0.4, 0, 0.2, 1], delay: 0.3 }}
   264→        />
   265→      </svg>
   266→      <div style={{
   267→        position: 'absolute', inset: 0,
   268→        display: 'flex', alignItems: 'center', justifyContent: 'center',
   269→      }}>
   270→        {children}
   271→      </div>
   272→    </div>
   273→  );
   274→}

   323→  totalCompleted,
   324→  totalTopics,
   325→  streak,
   326→  tasksCompleted,
   327→  tasksTotal,
   328→}: {
   329→  focusScore: number;
   330→  totalCompleted: number;
   331→  totalTopics: number;
   332→  streak: number;
   333→  tasksCompleted: number;
   334→  tasksTotal: number;

   336→  const [expandedCard, setExpandedCard] = useState<string | null>(null);
   337→
   338→  const progressPct = totalTopics > 0 ? Math.round((totalCompleted / totalTopics) * 100) : 0;
   339→
   340→  // Motivational messages based on state
   341→  const focusMessage = useMemo(() => {
   342→    if (tasksTotal === 0) return 'No tasks yet';
   343→    if (focusScore === 0) return 'Start your first task!';
   344→    if (focusScore < 50) return 'Getting warmer…';
   345→    if (focusScore < 100) return 'Keep going!';
   346→    return 'Perfect day! 🎉';
   347→  }, [focusScore, tasksTotal]);
   348→
   349→  const progressMessage = useMemo(() => {
   350→    if (totalCompleted === 0) return 'Your journey begins today';
   351→    if (progressPct < 10) return 'Great start!';
   352→    if (progressPct < 50) return 'Solid progress!';
   353→    return 'You\'re crushing it!';
   354→  }, [totalCompleted, progressPct]);
   355→
   356→  const streakMessage = useMemo(() => {
   357→    if (streak === 0) return 'Start your streak today';
   358→    if (streak === 1) return 'Day one — keep it up!';
   359→    if (streak < 7) return `${7 - streak} days until a week!`;
   360→    return 'On fire! 🔥';

   363→  const cards = [
   364→    {
   365→      id: 'focus',
   366→      label: 'Focus',
   367→      icon: <Target width={14} height={14} />,
   368→      color: '#FF3B30',
   369→      colorDim: 'rgba(255,59,48,0.12)',
   370→      ringProgress: focusScore,
   371→      ringColor: focusScore >= 100 ? '#22C55E' : focusScore >= 50 ? '#FF9500' : '#FF3B30',
   372→      mainValue: `${focusScore}%`,
   373→      subLabel: focusMessage,
   374→      detail: `Today: ${tasksCompleted}/${tasksTotal} tasks`,

   376→    {
   377→      id: 'progress',
   378→      label: 'Progress',
   379→      icon: <TrendingUp width={14} height={14} />,

   381→      colorDim: 'rgba(59,130,246,0.12)',
   382→      ringProgress: progressPct,
   383→      ringColor: progressPct >= 50 ? '#22C55E' : '#3B82F6',
   384→      mainValue: `${totalCompleted}`,
   385→      subLabel: progressMessage,
   386→      detail: `of ${totalTopics} topics · ${progressPct}%`,

   389→      id: 'streak',
   390→      label: 'Streak',

   397→      subLabel: streakMessage,
   398→      detail: 'days in a row',

   861→// FOCUS TIMER VIEW
   862→// ============================================
   863→function FocusTimerView() {
   864→  const [mode, setMode] = useState<'focus' | 'short' | 'long'>('focus');
   865→  const [totalSeconds, setTotalSeconds] = useState(25 * 60);
   866→  const [secondsLeft, setSecondsLeft] = useState(25 * 60);
   867→  const [isRunning, setIsRunning] = useState(false);
   868→  const [sessionsCompleted, setSessionsCompleted] = useState(0);
   869→
   870→  const modes = { focus: 25 * 60, short: 5 * 60, long: 15 * 60 };
   871→  const modeLabels = { focus: 'Focus 25m', short: 'Short 5m', long: 'Long 15m' };
   872→  const circumference = 2 * Math.PI * 120;
   873→  const progressPct = totalSeconds > 0 ? ((totalSeconds - secondsLeft) / totalSeconds) * 100 : 0;
   874→  const strokeDashoffset = circumference * (1 - progressPct / 100);
   875→
   876→  useEffect(() => {
   877→    if (!isRunning) return;
   878→    const interval = setInterval(() => {
   879→      setSecondsLeft(prev => {
   880→        if (prev <= 1) {
   881→          setIsRunning(false);
   882→          setSessionsCompleted(p => p + 1);
   883→          return 0;
   884→        }
   885→        return prev - 1;
   886→      });
   887→    }, 1000);
   888→    return () => clearInterval(interval);
   889→  }, [isRunning]);
   890→
   891→  const switchMode = (m: 'focus' | 'short' | 'long') => {
   892→    setMode(m);
   893→    setTotalSeconds(modes[m]);
   894→    setSecondsLeft(modes[m]);
   895→    setIsRunning(false);
   896→  };
   897→
   898→  const handleReset = () => {
   899→    setSecondsLeft(totalSeconds);
   900→    setIsRunning(false);
   901→  };
   902→
   903→  return (
   904→    <div className="view active">
   905→      <div className="focus-timer-container">
   906→        <div className="focus-mode-tabs">
   907→          {(Object.keys(modes) as Array<'focus' | 'short' | 'long'>).map(m => (
   908→            <button
   909→              key={m}
   910→              className={`focus-mode-tab ${mode === m ? 'active' : ''}`}
   911→              onClick={() => switchMode(m)}
   912→            >
   913→              {modeLabels[m]}
   914→            </button>
   915→          ))}
   916→        </div>
   917→
   918→        <div className="focus-session-label">
   919→          {mode === 'focus' ? 'FOCUS SESSION' : mode === 'short' ? 'SHORT BREAK' : 'LONG BREAK'}
   920→        </div>
   921→
   922→        <div className="focus-timer-circle">
   923→          <svg viewBox="0 0 260 260">
   924→            <circle className="track" cx="130" cy="130" r="120" />
   925→            <circle
   926→              className="progress"
   927→              cx="130" cy="130" r="120"
   928→              strokeDasharray={circumference}
   929→              strokeDashoffset={strokeDashoffset}
   930→            />
   931→          </svg>
   932→          <div className="focus-timer-display">
   933→            <div className="focus-timer-time">{formatTimer(secondsLeft)}</div>
   934→          </div>
   935→        </div>
   936→
   937→        <div className="focus-session-dots">
   938→          {[0, 1, 2, 3].map(i => (
   939→            <div
   940→              key={i}
   941→              className={`focus-dot ${i < sessionsCompleted ? 'completed' : ''} ${i === sessionsCompleted ? 'current' : ''}`}
   942→            />
   943→          ))}
   944→        </div>
   945→        <div className="focus-session-count">{sessionsCompleted} total</div>
   946→
   947→        <div className="focus-timer-controls">
   948→          <button
   949→            className="btn btn-accent"
   950→            onClick={() => setIsRunning(!isRunning)}
   951→          >
   952→            {isRunning ? <X width={15} height={15} /> : <Play width={15} height={15} />}
   953→            {isRunning ? 'Pause' : 'Start'}
   954→          </button>
   955→          <button className="btn btn-ghost" onClick={handleReset}>
   956→            <RotateCcw width={15} height={15} /> Reset
   957→          </button>
   958→        </div>
   959→      </div>
   960→    </div>
   961→  );
   962→}

  1338→  const [subjectDetailId, setSubjectDetailId] = useState<string | null>(null);
  1339→  const [initialFetchDone, setInitialFetchDone] = useState(false);
  1340→  const [showLoginFromLanding, setShowLoginFromLanding] = useState(false);
  1341→
  1342→  // All hooks must be called before any conditional returns
  1343→  const handleNewTask = useCallback(() => setShowModal(true), []);
  1344→
  1345→  // Combined view change handler — always clears subject detail so tabs work
  1346→  const handleViewChange = useCallback((v: ViewId) => {
  1347→    setCurrentView(v);
  1348→    setSubjectDetailId(null);
  1349→  }, []);
  1350→
  1351→  const handleFocusTimer = useCallback(() => handleViewChange('focus-timer'), [handleViewChange]);
  1352→
  1353→  const handleCourseClick = useCallback((subjectId: string) => {
  1354→    setSubjectDetailId(subjectId);
  1355→  }, []);
  1356→
  1357→  const closeSubjectDetail = store.closeSubjectDetail;
  1358→  const handleBackFromCourse = useCallback(() => {
  1359→    closeSubjectDetail();
  1360→    setSubjectDetailId(null);
  1361→  }, [closeSubjectDetail]);
  1362→
  1363→  // Fetch data when session is available
  1364→  const fetchData = store.fetchData;
  1365→  const initNetwork = store._initNetworkListeners;
  1366→  useEffect(() => {
  1367→    if (status === 'authenticated' && session?.user && !initialFetchDone) {
  1368→      const phone = (session.user as Record<string, unknown>).phone as string;
  1369→      if (phone) {
  1370→        fetchData(phone).then(() => setInitialFetchDone(true));
  1371→      }
  1372→    }
  1373→  }, [status, session, initialFetchDone, fetchData]);
  1374→
  1375→  // Initialize network listeners for offline/online detection
  1376→  useEffect(() => {
  1377→    const cleanup = initNetwork();
  1378→    return () => { cleanup?.(); };
  1379→  }, [initNetwork]);
  1380→
  1381→  // Show landing page if not authenticated
  1382→  // Also show landing page if session exists but has no user data (edge case)
  1383→  const hasValidSession = status === 'authenticated' && session?.user;
  1384→  if (!hasValidSession) {
  1385→    if (showLoginFromLanding) {
  1386→      return <LoginPage />;
  1387→    }
  1388→    return <LandingPage onGetStarted={() => setShowLoginFromLanding(true)} />;
  1389→  }
  1390→
  1391→  // Loading while checking auth
  1392→  if (status === 'loading') {
  1393→    return (
  1394→      <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'center', minHeight: '100vh', background: '#121212' }}>

  1399→      </div>
  1400→    );
  1401→  }
  1402→
  1403→  const studentName = store.student?.name || (session?.user?.name as string) || 'Student';
  1404→  const totalLecs = store.totalTopics;
  1405→
